    # helper for mse/brier selection
    def select_metric_winner(candidates, target, task):
        best_s = float('inf')
        best_k, best_p = None, None
        for k in candidates:
            if k not in combined_models: continue
            res = combined_models[k]
            pred = res['cate']
            valid = ~np.isnan(target) & ~np.isnan(pred)
            if not valid.any(): continue

            if task == 'regression':
                s = mean_squared_error(target[valid], pred[valid])
            else:
                s = brier_score_loss((target[valid]>0).astype(int), pred[valid])

            if s < best_s:
                best_s = s
                best_k = k
                best_p = pred
        return best_p, best_k
